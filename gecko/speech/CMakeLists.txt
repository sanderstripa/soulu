cmake_minimum_required(VERSION 3.24)
project(SouluSpeech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
foreach(EXT SSE42 AVX AVX2 AVX_VNNI BMI2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 OPENMP CUDA VULKAN)
 set(GGML_${EXT} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_Declare(soulu_whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG 4979e04f5dcaccb36057e059bbaed8a2f5288315 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(soulu_whisper)
add_executable(SouluSpeech main.cpp)
target_link_libraries(SouluSpeech PRIVATE whisper winmm bcrypt advapi32)
